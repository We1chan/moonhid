///|
fn installed_bytes(result : Result[Bytes, @moonhid.Diagnostic]) -> Bytes raise {
  match result {
    Ok(bytes) => bytes
    Err(error) => fail("\{error.code} @\{error.offset}: \{error.message}")
  }
}

///|
fn installed_layout(bytes : Bytes) -> @moonhid.Layout raise {
  match @moonhid.compile_descriptor(bytes) {
    Ok(layout) => layout
    Err(error) => fail(error.code)
  }
}

test "installed 0.4.0 templates, serialization, lint and JSON v2" {
  let keyboard = installed_bytes(@examples.keyboard_descriptor())
  assert_eq(
    keyboard,
    installed_bytes(
      @moonhid.parse_hex(
        "05 01 09 06 a1 01 05 07 19 e0 29 e7 15 00 25 01 75 01 95 08 81 02 75 08 95 01 81 01 19 00 29 65 15 00 25 65 75 08 95 06 81 00 05 08 19 01 29 05 15 00 25 01 75 01 95 05 91 02 75 03 95 01 91 01 c0",
      ),
    ),
  )
  for
    result in [
      @examples.mouse_descriptor(),
      @examples.keyboard_descriptor(),
      @examples.gamepad_descriptor(),
      @examples.consumer_descriptor(),
      @examples.vendor_descriptor(),
    ] {
    let bytes = installed_bytes(result)
    let items = match @moonhid.parse_items(bytes) {
      Ok(items) => items
      Err(error) => fail(error.code)
    }
    assert_eq(@moonhid.serialize_items(items), Ok(bytes))
    assert_eq(@moonhid.lint_descriptor(bytes), Ok([]))
    assert_true(
      @moonhid.descriptor_to_json(bytes)
      is Ok({ "schema_version": 2, "lints": [], .. }),
    )
  }
  let vendor = installed_layout(installed_bytes(@examples.vendor_descriptor()))
  assert_eq(@moonhid.report_length(vendor, @moonhid.Output, 0), Ok(64))
}

test "installed 0.4.0 report directory and keyboard Usage access" {
  let layout = installed_layout(
    installed_bytes(@examples.keyboard_descriptor()),
  )
  assert_eq(
    @moonhid.list_reports(layout),
    Ok([
      {
        kind: @moonhid.Input,
        report_id: 0,
        payload_bits: 64,
        byte_length: 8,
        data_count: 14,
      },
      {
        kind: @moonhid.Output,
        report_id: 0,
        payload_bits: 8,
        byte_length: 1,
        data_count: 5,
      },
    ]),
  )
  let idle = installed_bytes(@moonhid.new_report(layout, @moonhid.Input, 0))
  assert_eq(idle, b"\x00\x00\x00\x00\x00\x00\x00\x00")
  let pressed = installed_bytes(
    @moonhid.set_usages(layout, @moonhid.Input, idle, [
      { page: 7, id: 0xe1, },
      { page: 7, id: 4, },
      { page: 7, id: 6, },
    ]),
  )
  assert_eq(pressed, b"\x02\x00\x04\x06\x00\x00\x00\x00")
  assert_eq(
    @moonhid.active_usages(layout, @moonhid.Input, pressed),
    Ok([{ page: 7, id: 0xe1, }, { page: 7, id: 4, }, { page: 7, id: 6, }]),
  )
  assert_eq(
    @moonhid.clear_usages(layout, @moonhid.Input, pressed, [{ page: 7, id: 4, }]),
    Ok(b"\x02\x00\x00\x06\x00\x00\x00\x00"),
  )
  assert_eq(
    @moonhid.array_selector(layout.fields[2], { page: 7, id: 4, }),
    Ok(4L),
  )
}

test "installed 0.4.0 LED encoding and field edits preserve padding" {
  let layout = installed_layout(
    installed_bytes(@examples.keyboard_descriptor()),
  )
  assert_eq(
    @moonhid.encode_report(
      layout,
      @moonhid.Output,
      0,
      [1L, 1L, 0L, 0L, 0L].map(value => @moonhid.Integer(value)),
    ),
    Ok(b"\x03"),
  )
  let caps_lock = match
    @moonhid.find_fields(layout, @moonhid.Output, 0, { page: 8, id: 2, }) {
    Ok([reference]) => reference
    _ => fail("Caps Lock field missing")
  }
  assert_eq(
    @moonhid.read_field(
      layout,
      @moonhid.Output,
      b"\xa3",
      caps_lock.field_index,
      caps_lock.element_index,
    ),
    Ok(@moonhid.Integer(1L)),
  )
  assert_eq(
    @moonhid.write_field(
      layout,
      @moonhid.Output,
      b"\xa3",
      caps_lock.field_index,
      caps_lock.element_index,
      @moonhid.Integer(0L),
    ),
    Ok(b"\xa1"),
  )
  assert_eq(
    @moonhid.set_usage_value(
      layout,
      @moonhid.Output,
      b"\xa1",
      { page: 8, id: 3, },
      @moonhid.Integer(1L),
    ),
    Ok(b"\xa5"),
  )
  assert_eq(
    @moonhid.get_usage_value(layout, @moonhid.Output, b"\xa5", {
      page: 8,
      id: 3,
    }),
    Ok(@moonhid.Integer(1L)),
  )
}

test "installed 0.4.0 three report directions and multiple IDs" {
  let layout = installed_layout(
    installed_bytes(
      @moonhid.parse_hex(
        "85 01 15 00 26 ff 00 75 08 95 01 81 02 91 02 b1 02 85 02 75 10 26 ff ff 81 02",
      ),
    ),
  )
  for kind in [@moonhid.Input, @moonhid.Output, @moonhid.Feature] {
    assert_eq(
      @moonhid.encode_report(layout, kind, 1, [@moonhid.Integer(255L)]),
      Ok(b"\x01\xff"),
    )
    let report = match @moonhid.decode_report(layout, kind, b"\x01\xff") {
      Ok(report) => report
      Err(error) => fail(error.code)
    }
    assert_eq(report.values[0].value, 255L)
  }
  assert_eq(
    @moonhid.encode_report(layout, @moonhid.Input, 2, [@moonhid.Integer(65535L)]),
    Ok(b"\x02\xff\xff"),
  )
  assert_true(
    @moonhid.encode_report(layout, @moonhid.Output, 2, [])
    is Err({ code: "unknown_report", .. }),
  )
  assert_true(
    @moonhid.encode_report(layout, @moonhid.Input, 1, [@moonhid.Integer(256L)])
    is Err({ code: "value_range", .. }),
  )
}

test "installed 0.4.0 builder, Null initialization and physical angles" {
  let descriptor = installed_bytes(
    @moonhid.DescriptorBuilder::new()
    .usage_page(1)
    .usage(5)
    .collection(1)
    .usage(0x39)
    .logical_range(0L, 7L)
    .physical_range(0L, 315L)
    .unit(0x14)
    .report_size(4)
    .report_count(1)
    .input(0x42)
    .input(1)
    .end_collection()
    .build(),
  )
  let layout = installed_layout(descriptor)
  let idle = installed_bytes(@moonhid.new_report(layout, @moonhid.Input, 0))
  assert_eq(idle, b"\x0f")
  assert_true(
    @moonhid.get_physical_value(layout, @moonhid.Input, idle, {
      page: 1,
      id: 0x39,
    })
    is Err({ code: "null_value", .. }),
  )
  let east = installed_bytes(
    @moonhid.set_physical_value(
      layout,
      @moonhid.Input,
      idle,
      { page: 1, id: 0x39, },
      90.0,
    ),
  )
  assert_eq(east, b"\x02")
  assert_eq(
    @moonhid.get_physical_value(layout, @moonhid.Input, east, {
      page: 1,
      id: 0x39,
    }),
    Ok(90.0),
  )
  assert_eq(@moonhid.physical_value(layout.fields[0], 2L), Ok(90.0))
  assert_eq(@moonhid.logical_from_physical(layout.fields[0], 90.0), Ok(2L))
}

test "installed 0.4.0 wide Raw Feature report" {
  let descriptor = installed_bytes(
    @moonhid.DescriptorBuilder::new()
    .usage_page(0xff00)
    .usage(1)
    .collection(1)
    .report_id(7)
    .usage(2)
    .logical_range(0L, 1L)
    .report_size(40)
    .report_count(1)
    .feature(2)
    .end_collection()
    .build(),
  )
  let layout = installed_layout(descriptor)
  let raw = b"\xff\xff\xff\xff\x01"
  let wire = installed_bytes(
    @moonhid.encode_report(layout, @moonhid.Feature, 7, [@moonhid.Raw(raw)]),
  )
  assert_eq(wire, b"\x07\xff\xff\xff\xff\x01")
  assert_eq(
    @moonhid.read_field(layout, @moonhid.Feature, wire, 0, 0),
    Ok(@moonhid.Raw(raw)),
  )
  let report = match @moonhid.decode_report(layout, @moonhid.Feature, wire) {
    Ok(report) => report
    Err(error) => fail(error.code)
  }
  assert_eq(report.opaque_values[0].bytes, raw)
  assert_true(
    @moonhid.encode_report(layout, @moonhid.Feature, 7, [@moonhid.Integer(1L)])
    is Err({ code: "value_type", .. }),
  )
}
