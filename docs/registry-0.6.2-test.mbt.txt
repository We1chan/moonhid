///|
fn installed_bytes(result : Result[Bytes, @moonhid.Diagnostic]) -> Bytes raise {
  match result {
    Ok(bytes) => bytes
    Err(error) => fail("\{error.code} @\{error.offset}: \{error.message}")
  }
}

///|
fn installed_layout(bytes : Bytes) -> @moonhid.Layout raise {
  match @moonhid.compile_descriptor(bytes) {
    Ok(layout) => layout
    Err(error) => fail(error.code)
  }
}

///|
fn installed_report(
  layout : @moonhid.Layout,
  kind : @moonhid.ReportKind,
  wire : Bytes,
) -> @moonhid.DecodedReport raise {
  match @moonhid.decode_report(layout, kind, wire) {
    Ok(report) => report
    Err(error) => fail(error.code)
  }
}

test "installed 0.6.1 Buffered Bytes in all directions and JSON" {
  let descriptor = installed_bytes(
    @moonhid.parse_hex(
      "06 00 ff 09 01 a1 01 85 05 15 00 25 01 75 08 95 04 09 02 82 42 01 09 03 92 42 01 09 04 b2 42 01 c0",
    ),
  )
  let layout = installed_layout(descriptor)
  let raw = b"\x00\xff\x4d\x4e"
  let wire = b"\x05\x00\xff\x4d\x4e"
  assert_eq(@moonhid.lint_descriptor(descriptor), Ok([]))
  for
    (index, kind) in [
      (0, @moonhid.Input),
      (1, @moonhid.Output),
      (2, @moonhid.Feature),
    ] {
    assert_eq(
      @moonhid.encode_report(layout, kind, 5, [@moonhid.Raw(raw)]),
      Ok(wire),
    )
    assert_eq(@moonhid.new_report(layout, kind, 5), Ok(b"\x05\x00\x00\x00\x00"))
    let report = installed_report(layout, kind, wire)
    assert_eq(report.values, [])
    assert_eq(report.opaque_values, [])
    assert_eq(report.buffered_values, [
      { field_index: index, bit_offset: 0, bytes: raw, },
    ])
    assert_true(
      @moonhid.report_to_json(report)
      is {
        "values": [],
        "opaque_values": [],
        "buffered_values": [{ "hex": "00 ff 4d 4e", "bit_offset": 0, .. }],
        ..
      },
    )
  }
  match @moonhid.list_reports(layout) {
    Ok(reports) =>
      assert_eq(
        reports.map(r => (r.payload_bits, r.byte_length, r.data_count)),
        [(32, 5, 1), (32, 5, 1), (32, 5, 1)],
      )
    Err(error) => fail(error.code)
  }
}

test "installed 0.6.1 buffered edits preserve neighboring bits and Usage access" {
  let layout = installed_layout(
    installed_bytes(
      @moonhid.parse_hex(
        "06 00 ff 85 07 15 00 25 01 75 03 95 01 81 02 75 05 81 01 75 08 95 02 09 02 82 42 01 95 01 81 02",
      ),
    ),
  )
  let wire = b"\x07\xf9\x00\xff\x01"
  let usage : @moonhid.Usage = { page: 65280, id: 2, }
  assert_eq(
    @moonhid.read_field(layout, @moonhid.Input, wire, 2, 0),
    Ok(@moonhid.Raw(b"\x00\xff")),
  )
  assert_eq(
    @moonhid.write_field(
      layout,
      @moonhid.Input,
      wire,
      2,
      0,
      @moonhid.Raw(b"OK"),
    ),
    Ok(b"\x07\xf9OK\x01"),
  )
  assert_eq(
    @moonhid.get_usage_value(layout, @moonhid.Input, wire, usage),
    Ok(@moonhid.Raw(b"\x00\xff")),
  )
  assert_eq(
    @moonhid.set_usage_value(
      layout,
      @moonhid.Input,
      wire,
      usage,
      @moonhid.Raw(b"OK"),
    ),
    Ok(b"\x07\xf9OK\x01"),
  )
  assert_true(
    @moonhid.read_field(layout, @moonhid.Input, wire, 2, 1)
    is Err({ code: "element_index", .. }),
  )
  assert_true(
    @moonhid.write_field(
      layout,
      @moonhid.Input,
      wire,
      2,
      0,
      @moonhid.Integer(1L),
    )
    is Err({ code: "value_type", .. }),
  )
  assert_true(
    @moonhid.write_field(layout, @moonhid.Input, wire, 2, 0, @moonhid.Raw(b"X"))
    is Err({ code: "value_range", .. }),
  )
  assert_true(
    @moonhid.physical_value(layout.fields[2], 1L)
    is Err({ code: "value_type", .. }),
  )
  assert_eq(
    installed_report(layout, @moonhid.Input, wire).values.map(v => v.value),
    [1L, 1L],
  )
}

test "installed 0.6.1 buffered widths alignment and resource limits" {
  for (size, count) in [(1, 16), (4, 4), (16, 1), (40, 2)] {
    let descriptor = installed_bytes(
      @moonhid.DescriptorBuilder::new()
      .report_size(size)
      .report_count(count)
      .input(256)
      .build(),
    )
    let layout = installed_layout(descriptor)
    let raw = Bytes::makei(size * count / 8, i => (255 - i).to_byte())
    assert_eq(
      @moonhid.encode_report(layout, @moonhid.Input, 0, [@moonhid.Raw(raw)]),
      Ok(raw),
    )
    assert_eq(
      installed_report(layout, @moonhid.Input, raw).buffered_values[0].bytes,
      raw,
    )
  }
  for
    bytes in [
      b"\x75\x01\x95\x01\x82\x02\x01", b"\x75\x01\x95\x01\x81\x01\x75\x08\x82\x02\x01",
    ] {
    assert_true(
      @moonhid.compile_descriptor(bytes)
      is Err({ code: "buffered_alignment", .. }),
    )
  }
  assert_true(
    @moonhid.compile_descriptor(b"\x75\x08\x95\x01\x82\x02\x03")
    is Err({ code: "main_flags", .. }),
  )
  let layout = installed_layout(b"\x85\x07\x75\x08\x96\x00\x20\x82\x02\x01")
  let raw = Bytes::makei(8192, i => (i % 256).to_byte())
  let wire = installed_bytes(
    @moonhid.encode_report(layout, @moonhid.Input, 7, [@moonhid.Raw(raw)]),
  )
  assert_eq(wire.length(), 8193)
  assert_eq(
    installed_report(layout, @moonhid.Input, wire).buffered_values[0].bytes,
    raw,
  )
  assert_true(
    @moonhid.compile_descriptor(b"\x75\x08\x96\x01\x20\x82\x02\x01")
    is Err({ code: "report_limit", .. }),
  )
}

  let east = installed_bytes(
    @moonhid.set_physical_value(

  )
  assert_eq(east, b"\x02")
  assert_eq(
    @moonhid.get_physical_value(layout, @moonhid.Input, east, {
      page: 1,

    Ok(90.0),
